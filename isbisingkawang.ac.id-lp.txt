<script>
(function () {
  const wrap = document.getElementById("rtpCalendarlpx");
  const clock = document.getElementById("rtpClocklpx");

  if (!wrap || !clock) return;

  const dayNames = ["MIN", "SEN", "SEL", "RAB", "KAM", "JUM", "SAB"];
  const monthNames = ["Jan", "Feb", "Mar", "Apr", "Mei", "Jun", "Jul", "Agu", "Sep", "Okt", "Nov", "Des"];

  function seededRandom(seed) {
    let x = Math.sin(seed) * 10000;
    return x - Math.floor(x);
  }

  function getRtpValue(date, index) {
    const now = new Date();
    const updateBlock = Math.floor(now.getMinutes() / 15);
    const seed =
      date.getFullYear() * 10000 +
      (date.getMonth() + 1) * 100 +
      date.getDate() +
      index * 77 +
      updateBlock * 33;

    const min = 85;
    const max = 98.8;
    const value = min + seededRandom(seed) * (max - min);
    return value.toFixed(2);
  }

  function getStatus(rtp) {
    const value = parseFloat(rtp);

    if (value >= 98) return "HOT";
    if (value >= 95) return "STABIL";
    return "NORMAL";
  }

  function renderCalendar() {
    const today = new Date();
    const currentDay = today.getDay();

    wrap.innerHTML = "";

    for (let i = 0; i < 7; i++) {
      const date = new Date();
      date.setDate(today.getDate() + i);

      const rtp = getRtpValue(date, i);
      const status = getStatus(rtp);

      const card = document.createElement("div");
      card.className = "rtp-day-card" + (i === 0 ? " active" : "");

      card.innerHTML = `
        <div class="rtp-day-name">${i === 0 ? "HARI INI" : dayNames[date.getDay()]}</div>
        <div class="rtp-day-date">${date.getDate()} ${monthNames[date.getMonth()]}</div>
        <div class="rtp-percent">${rtp}%</div>
        <div class="rtp-status">${status}</div>
      `;

      wrap.appendChild(card);
    }
  }

  function updateClock() {
    const now = new Date();

    const jam = String(now.getHours()).padStart(2, "0");
    const menit = String(now.getMinutes()).padStart(2, "0");
    const detik = String(now.getSeconds()).padStart(2, "0");

    const nextMinute = 15 - (now.getMinutes() % 15);
    const nextSecond = 60 - now.getSeconds();

    clock.textContent = `LIVE ${jam}:${menit}:${detik} | Update RTP tiap 15 menit`;
  }

  renderCalendar();
  updateClock();

  setInterval(updateClock, 1000);

  setInterval(function () {
    renderCalendar();
  }, 60000);
})();
</script>

<script>
(() => {
  const rainCanvas = document.getElementById("rain-canvas");
  const rainCtx = rainCanvas.getContext("2d");

  const flash = document.getElementById("lightning-flash");

  const lightningCanvas = document.getElementById("lightning-canvas");
  const lightningCtx = lightningCanvas.getContext("2d");

  function random(min, max){
    return Math.random() * (max - min) + min;
  }

  function resizeCanvas(){
    rainCanvas.width = window.innerWidth;
    rainCanvas.height = window.innerHeight;

    lightningCanvas.width = window.innerWidth;
    lightningCanvas.height = window.innerHeight;
  }

  resizeCanvas();
  window.addEventListener("resize", resizeCanvas);

  const drops = [];
  const dropCount = Math.min(420, window.innerWidth * 0.7);

  for(let i = 0; i < dropCount; i++){
    drops.push({
      x: random(0, rainCanvas.width),
      y: random(0, rainCanvas.height),
      len: random(16, 32),
      speed: random(10, 20),
      width: random(0.9, 1.5),
      alpha: random(0.28, 0.55)
    });
  }

  function drawRain(){
    rainCtx.clearRect(0, 0, rainCanvas.width, rainCanvas.height);

    for(let d of drops){
      rainCtx.beginPath();
      rainCtx.strokeStyle = `rgba(255,255,255,${d.alpha})`;
      rainCtx.lineWidth = d.width;
      rainCtx.moveTo(d.x, d.y);
      rainCtx.lineTo(d.x, d.y + d.len);
      rainCtx.stroke();

      d.y += d.speed;

      if(d.y > rainCanvas.height){
        d.y = -30;
        d.x = random(0, rainCanvas.width);
      }
    }

    requestAnimationFrame(drawRain);
  }

  drawRain();

  function drawLightning(){
    lightningCtx.clearRect(0, 0, lightningCanvas.width, lightningCanvas.height);

    let x = random(lightningCanvas.width * 0.15, lightningCanvas.width * 0.85);
    let y = 0;

    lightningCtx.beginPath();
    lightningCtx.moveTo(x, y);

    while(y < lightningCanvas.height * 0.65){
      x += random(-45, 45);
      y += random(18, 42);
      lightningCtx.lineTo(x, y);

      if(Math.random() > 0.72){
        const branchX = x + random(-120, 120);
        const branchY = y + random(35, 90);

        lightningCtx.moveTo(x, y);
        lightningCtx.lineTo(branchX, branchY);
        lightningCtx.moveTo(x, y);
      }
    }

    lightningCtx.strokeStyle = "rgba(255,255,255,0.95)";
    lightningCtx.lineWidth = random(2, 4);
    lightningCtx.shadowBlur = 35;
    lightningCtx.shadowColor = "#7fd3ff";
    lightningCtx.stroke();

    flash.style.transition = "none";
    flash.style.opacity = random(0.28, 0.48);

    setTimeout(() => {
      flash.style.transition = "opacity 0.35s ease-out";
      flash.style.opacity = 0;
      lightningCtx.clearRect(0, 0, lightningCanvas.width, lightningCanvas.height);
    }, 160);
  }

  function dramaticLightning(){
    drawLightning();

    if(Math.random() > 0.45){
      setTimeout(drawLightning, 130);
    }

    if(Math.random() > 0.65){
      setTimeout(drawLightning, 280);
    }

    setTimeout(dramaticLightning, random(6500, 13000));
  }

  setTimeout(dramaticLightning, 3500);
})();
</script>
